// Pixie Deck - step 1: read the keys, say what was pressed.
//
// Hardware: XIAO ESP32S3 + SH1106 128x64 mono OLED (I2C) + 8 keys in a 2x4
// matrix.
//
// The firmware's whole job is to report what happened. It knows nothing about
// tmux, sessions, or your Mac - it just says "FOX was pressed" and lets the
// host decide what that means. Keeping that boundary is what makes the device
// testable on its own and replaceable later.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SH110X.h>

#define OLED_ADDR 0x3C
#define SDA_PIN   D4
#define SCL_PIN   D5

Adafruit_SH1106G display(128, 64, &Wire, -1);

// ---------------------------------------------------------------------------
// The key matrix
// ---------------------------------------------------------------------------
// 8 keys would need 8 pins wired one-per-key. A matrix gets it down to 6: the
// keys sit at the crossings of 2 rows and 4 columns, and we find which one is
// pressed by energising one row at a time and seeing which columns respond.
//
// The column order below was established by pressing each key in turn and
// reading back what the board reported - NOT by reading the wiring diagram,
// which turned out to be reversed. On this project, measure.

const uint8_t ROW_PINS[2] = { D0, D1 };
const uint8_t COL_PINS[4] = { D7, D8, D9, D10 };

// Index = row * 4 + col, so this array is the physical layout, read left to
// right, top to bottom. Rewiring a key means editing this one line.
const char *KEY_NAMES[8] = {
  "TERM",  "MIC",    "FOX", "PNDA",
  "ENTER", "CANCEL", "CAT", "OWL"
};

bool keyDown[8]           = { false };
unsigned long keyChanged[8] = { 0 };

// A mechanical switch does not close cleanly - the contacts bounce for a few
// milliseconds, which reads as several presses. Ignoring further changes for
// 25ms after any change is the standard fix.
const unsigned long DEBOUNCE_MS = 25;

String lastKey = "";

// ---------------------------------------------------------------------------

void showLastKey() {
  display.clearDisplay();
  display.setTextColor(SH110X_WHITE);

  display.setTextSize(1);
  display.setCursor(0, 0);
  display.print("Pixie Deck");

  display.setTextSize(2);
  display.setCursor(0, 28);
  display.print(lastKey.length() ? lastKey : "...");

  display.display();
}

void scanMatrix() {
  for (uint8_t r = 0; r < 2; r++) {
    // Rows are OUTPUT_OPEN_DRAIN: driving LOW pulls the line down, but HIGH
    // just releases it rather than actively pushing 3.3V out. That way, if two
    // keys in the same column are held at once, one row cannot fight the
    // other and short them together.
    digitalWrite(ROW_PINS[r], LOW);
    delayMicroseconds(30);            // let the line settle before reading

    for (uint8_t c = 0; c < 4; c++) {
      uint8_t idx = r * 4 + c;

      // Columns are INPUT_PULLUP, so they idle HIGH. A pressed key connects
      // the column to the row we just pulled LOW - so LOW means pressed.
      bool pressed = (digitalRead(COL_PINS[c]) == LOW);

      if (pressed != keyDown[idx] && millis() - keyChanged[idx] > DEBOUNCE_MS) {
        keyDown[idx]    = pressed;
        keyChanged[idx] = millis();

        if (pressed) {
          // One line per event. Newline-terminated so the host can read it
          // with a plain `read` and never has to parse a stream.
          Serial.println(KEY_NAMES[idx]);
          lastKey = KEY_NAMES[idx];
          showLastKey();
        }
      }
    }

    digitalWrite(ROW_PINS[r], HIGH);  // release before testing the next row
  }
}

void setup() {
  Serial.begin(115200);

  for (uint8_t r = 0; r < 2; r++) {
    pinMode(ROW_PINS[r], OUTPUT_OPEN_DRAIN);
    digitalWrite(ROW_PINS[r], HIGH);
  }
  for (uint8_t c = 0; c < 4; c++) {
    pinMode(COL_PINS[c], INPUT_PULLUP);
  }

  Wire.begin(SDA_PIN, SCL_PIN);
  display.begin(OLED_ADDR, true);
  display.setRotation(0);
  showLastKey();
}

void loop() {
  scanMatrix();
  delay(2);   // 500 scans a second is far faster than any finger
}
